cmake_minimum_required(VERSION 3.16)
project(TelegramPolice CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(Td REQUIRED)
add_executable(TelegramPolice WIN32 src/main.cpp src/tdclient.cpp src/mainwindow.cpp src/qrcodegen.cpp res/res.qrc res/app.rc)
target_link_libraries(TelegramPolice PRIVATE Qt6::Widgets Td::TdJson $<$<BOOL:${WIN32}>:dwmapi>)
